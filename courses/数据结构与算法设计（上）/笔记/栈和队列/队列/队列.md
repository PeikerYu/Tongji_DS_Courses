**链栈与顺序栈的选择**
| 栈类型 | 存储结构 | 特点           | 适合场景    |
| --- | ---- | ------------ | ------- |
| 顺序栈 | 数组   | 快速，容易实现，容量固定 | 元素上限已知  |
| 链栈  | 链表   | 动态扩容，不会溢出    | 不知道最大容量 |

顺序队列及其常用函数
注意，rear是虚指，front是实指，rear指向的是队尾有效元素的下一个位置，这样便于把循环队列判空和判满的两种情况分开
struct Queue
{
    int data[MAX_SIZE];
    int front;
    int rear;
};

void InitQueue(Queue *q)
{
    q->front=q->rear=0;
}

int LengthQueue(Queue *q)
{
    return (Q.rear-Q.front+MAX_SIZE)%MAX_SIZE;
}

bool isEmpty(Queue *q)
{
    return q->front==q->rear;
}

bool isFull(Queue *q)
{
    return (q->rear+1)%MAX_SIZE==q->front;
}

int Push(Queue *q,int val)
{
    if(isFull(q))   
        return -1;
    q->data[q->rear]=val;
    q->rear=(q->rear+1)%MAX_SIZE;
}

int Pop(Queue*q)
{
    if(isEmpty(q))  
        return -1;
    int temp=q->data[q->front];
    q->front=(q->front+1)%MAX_SIZE;
    return temp;
}

int getFront(Queue *q)
{
    if(isEmpty) 
        return INF;
    int e=q->data[q->front];
    return e;
}

int getRear(Queue*q)
{
    if(isEmpty(q))  
        return INF;
    return q->data[(q->rear - 1 + MAXSIZE) % MAXSIZE]; 
}

链式队列及其常用函数（带头节点版）
实际上，链式队列比带头结点的单链表只是多了一个尾结点（尾指针）
struct QNode
{
    int data;
    QNode *next;
};

struct Queue
{
    QNode *front; //保存队头元素
    QNode *rear; //队尾
};

void InitQueue(Queue &Q)
{
    Q.front=Q.rear=(QNode*)malloc(sizeof(QNode));
    if(Q.front==NULL) exit(1);
    Q.front->next=NULL;
}

bool InQueue(Queue &Q,int e)
{
    QNode *p=(QNode*)malloc(sizeof(QNode));
    p->data=e;
    p->next=NULL; //由于队列的插入在表尾，因此新节点的next必然是null
    Q.rear->next=p;
    Q.rear=p;
    return true;
}

bool OutQueue(Queue &Q,int &e)
{
    if(IsEmpty)
        return false;
    QNode *p=Q.front->next; //p 指向队头元素的结点（注意：Q.front 是头结点，它不存数据）
    e=p->data;
    Q.front->next=p->next;
    if(p==Q.rear)
        Q.rear=Q.front;
        // 如果删除的是队尾结点，则需要更新 Q.rear 指向头结点
    free(p);
    return true;
}

bool IsEmpty(Queue &Q)
{
    if(Q.rear==Q.front)
        return true;
    else
        return false;
}
